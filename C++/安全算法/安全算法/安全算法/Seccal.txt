//种子得到校验码   dwSeed:27回复的内容  dwMask:掩码，每个控制器对应一个掩码
DWORD SeedToKey(UINT dwSeed, UINT dwMask)
{
	DWORD dwKey = 0;

	BYTE seed[4];
	BYTE Arrayxor[4];

	seed[0] = (dwSeed >> 24) & 0xff;
	seed[1] = (dwSeed >> 16) & 0xff;
	seed[2] = (dwSeed >> 8) & 0xff;
	seed[3] = (dwSeed >> 0) & 0xff;

	Arrayxor[3] = (dwMask >> 24) & 0xff;
	Arrayxor[2] = (dwMask >> 16) & 0xff;
	Arrayxor[1] = (dwMask >> 8) & 0xff;
	Arrayxor[0] = (dwMask >> 0) & 0xff;

	BYTE val[4] = { 0 };
	BYTE valKey[4] = { 0 };
	val[0] = seed[0] ^ Arrayxor[0];//27回复的4个字节，分别与掩码字节异或：相同为0，不同为1
	val[1] = seed[1] ^ Arrayxor[1];
	val[2] = seed[2] ^ Arrayxor[2];
	val[3] = seed[3] ^ Arrayxor[3];

	valKey[0] = ((val[3] & 0xF0) >> 4) | (16 * (val[0] & 0xF));
	valKey[1] = ((val[2] & 0xF0) >> 4) | (16 * (val[1] & 0xF));
	valKey[2] = ((val[0] & 0x3E) >> 1) | (8 * (val[3] & 0x1C));
	valKey[3] = ((val[1] & 0xF) >> 4) | (val[2] & 0xF0);

	dwKey = (valKey[0] << 24) + (valKey[1] << 16) + (valKey[2] << 8) + valKey[3];

	return dwKey;
}

DWORD SeedToKey_Level2(UINT dwSeed, UINT dwMask)
{
	DWORD dwKey = 0;
	BYTE seed[4];
	BYTE Arrayxor[4];
	seed[0] = (dwSeed >> 24) & 0xff;
	seed[1] = (dwSeed >> 16) & 0xff;
	seed[2] = (dwSeed >> 8) & 0xff;
	seed[3] = (dwSeed >> 0) & 0xff;
	Arrayxor[3] = (dwMask >> 24) & 0xff;
	Arrayxor[2] = (dwMask >> 16) & 0xff;
	Arrayxor[1] = (dwMask >> 8) & 0xff;
	Arrayxor[0] = (dwMask >> 0) & 0xff;
	BYTE val[4] = { 0 };
	BYTE valKey[4] = { 0 };
	for (int i = 0; i < 4; i++)
	{
		if (i == 0)
		{
			Arrayxor[i] = Arrayxor[i] ^ 0x56;
		}
		else
		{
			Arrayxor[i] = Arrayxor[i-1] + 2 * i;
		}
		val[i] = seed[i] ^ Arrayxor[i];
	}
	valKey[0] = ((val[3] & 0xF0) >> 4) | (16 * (val[0] & 0x37));
	valKey[1] = ((val[2] & 0xF0) >> 4) | (4 * (val[1] & 0xF));
	valKey[2] = ((val[2] & 0x3E) >> 1) | (8 * (val[3] & 0x1C));
	valKey[3] = ((val[1] & 0x2F) >> 4) | (val[2] & 0xF0);
	dwKey = (valKey[0] << 24) + (valKey[1] << 16) + (valKey[2] << 8) + valKey[3];
	return dwKey;
}
DWORD SeedToKey_FBL(UINT dwSeed, UINT dwMask)
{
	DWORD dwKey = 0;

	BYTE seed[4];
	BYTE Arrayxor[4];

	seed[0] = (dwSeed >> 24) & 0xff;
	seed[1] = (dwSeed >> 16) & 0xff;
	seed[2] = (dwSeed >> 8) & 0xff;
	seed[3] = (dwSeed >> 0) & 0xff;

	Arrayxor[3] = (dwMask >> 24) & 0xff;
	Arrayxor[2] = (dwMask >> 16) & 0xff;
	Arrayxor[1] = (dwMask >> 8) & 0xff;
	Arrayxor[0] = (dwMask >> 0) & 0xff;

	BYTE val[4] = { 0 };
	BYTE valKey[4] = { 0 };
	val[0] = seed[0] ^ Arrayxor[0];
	val[1] = seed[1] ^ Arrayxor[1];
	val[2] = seed[2] ^ Arrayxor[2];
	val[3] = seed[3] ^ Arrayxor[3];

	valKey[0] = ((val[3] & 0x3C) >> 2) | val[1] & 0xF0;
	valKey[1] = val[2] & 0xF | val[0] & 0xF0;
	valKey[2] = ((val[1] & 0x3C) >> 2) | 16 * (val[3] & 0xF);
	valKey[3] = ((val[0] & 0x78) >> 3) | 8 * (val[2] & 0x1E);

	dwKey = (valKey[0] << 24) + (valKey[1] << 16) + (valKey[2] << 8) + valKey[3];

	return dwKey;
}

/*******************************/
unsigned long Nexteer_croleft(unsigned long c, unsigned int b)
{
	unsigned long left = c << b;
	unsigned long right = c >> (32 - b);
	unsigned long croleftvalue = left | right;
	return croleftvalue;
}
/*******************************/

/*******************************/
unsigned short int Nexteer_croshortright(unsigned short int c, unsigned short int b)
{
	unsigned short int right = c >> b;
	unsigned short int left = c << (16 - b);
	unsigned short int crorightvalue = left | right;
	return crorightvalue;
}
/*******************************/

/*******************************/
unsigned long Nexteer_mulu32(unsigned long val1, unsigned long val2)
{
	unsigned long x, y, z, p;
	x = (val1&NEXTEER_UDS_KEYMASK) | ((~val1)&val2);
	y = ((Nexteer_croleft(val1, 1))&(Nexteer_croleft(val2, 14))) | ((Nexteer_croleft(NEXTEER_UDS_KEYMASK, 21))&(~(Nexteer_croleft(val1, 30))));
	z = (Nexteer_croleft(val1, 17)) ^ (Nexteer_croleft(val2, 4)) ^ (Nexteer_croleft(NEXTEER_UDS_KEYMASK, 11));
	p = x ^ y^z;
	return p;
}
/*******************************/
unsigned long Nexteer_uds_calc_key(unsigned long seed)
{
	unsigned long temp;
	unsigned short int  index;
	unsigned short int  mult1;
	unsigned short int  mult2;

	if (seed == 0)
	{
		seed = NEXTEER_DEFAULT_SEED;
	}
	else
	{
	}
	for (index = 0x5D39, temp = 0x80000000; temp; temp >>= 1)
	{
		if (temp & seed)
		{
			index = Nexteer_croshortright(index, 1);
			if (temp & NEXTEER_UDS_KEYMASK)
			{
				index ^= 0x74c9;
			}
		}
	}
	mult1 = (Nexteer_uds_keymul[(index >> 2) & ((1 << 5) - 1)] ^ index);
	mult2 = (Nexteer_uds_keymul[(index >> 8) & ((1 << 5) - 1)] ^ index);
	temp = (((unsigned long)mult1) << 16) | ((unsigned long)mult2);
	temp = Nexteer_mulu32(seed, temp);
	return temp;
}
BOOL Security_Nexteer_EPS(int iLevel)
{
	BOOL	bFlag = TRUE;

	DWORD   uiSEED = 0;
	DWORD   uiMask = 0;
	DWORD   uiKEY = 0;
	CBinary binReciver;

	CBinary bin10("\x10\x03", 2);
	CBinary bin2701("\x27\x01", 2);
	CBinary bin2702("\x27\x02\x00\x00\x00\x00", 6);


	switch (iLevel)
	{
	case 1: //扩展诊断会话
		bin10.SetAt(1, 0x03);
		bin2701.SetAt(1, 0x01);
		bin2702.SetAt(1, 0x02);
		break;
	case 2: //LEVEL2 编程会话
		bin10.SetAt(1, 0x02);//10 02
		bin2701.SetAt(1, 0x09);//27 09 
		bin2702.SetAt(1, 0x0A);//27 0A 00 00 00 00
		break;
	default:
		break;
	}

	StatusBox(TEXT_SECURITY_ING);//去掉“TEXT_”后，可以在TEXT文本中查询到

	CCommEcu::SetTimeout(15000);
	CCommEcu::SetCommTime(5, 1, 10000, 50);	//njb 20190620 修改
	binReciver = SendReceiveFrame(g_set.default_flag, bin10);
	if ((binReciver.IsEmpty()) || (binReciver[0] == 0x7F))
	{
		MessageBox(TEXT_SECURITY_FAILED);
		return FALSE;
	}

	//延时1s再执行27解锁
	Delay(1000);

	binReciver = SendReceiveFrame(g_set.default_flag, bin2701);

	if (binReciver.IsEmpty())
	{
		MessageBox(TEXT_COMM_FAIL);
		return FALSE;
	}

	if (binReciver[0] == 0x7F)
	{
		NegativeResponse_0x7F(binReciver);
		return FALSE;
	}

	if (binReciver.GetSize() < 6)
	{
		MessageBox(TEXT_SECURITY_FAILED);
		return false;
	}

	uiSEED = (binReciver[2] << 24) + (binReciver[3] << 16) + (binReciver[4] << 8) + binReciver[5];//回复的值是不固定的
	if ((uiSEED == 0xFFFFFFFF))//不需要安全访问
	{
		return TRUE;
	}

	uiKEY = Nexteer_uds_calc_key(uiSEED);

	bin2702[2] = (BYTE)((uiKEY & 0xFF000000) >> 24);
	bin2702[3] = (BYTE)((uiKEY & 0x00FF0000) >> 16);
	bin2702[4] = (BYTE)((uiKEY & 0x0000FF00) >> 8);
	bin2702[5] = (BYTE)((uiKEY & 0x000000FF));

	binReciver = SendReceiveFrame(g_set.default_flag, bin2702);
	Delay(500);

	if (binReciver.IsEmpty())
	{
		MessageBox(TEXT_COMM_FAIL);
		return FALSE;
	}

	if ((binReciver[0] == 0x7F))
	{
		NegativeResponse_0x7F(binReciver);
		MessageBox(TEXT_SECURITY_FAILED);
		return FALSE;
	}

	return bFlag;
}

{
	BOOL	bFlag = TRUE;
	DWORD   uiSEED = 0;
	DWORD   uiMask = 0;
	DWORD   uiKEY = 0;
	CBinary binReciver;

	CBinary bin10("\x10\x03", 2);
	CBinary bin2701("\x27\x01", 2);
	CBinary bin2702("\x27\x02\x00\x00\x00\x00", 6);

	switch (iLevel)
	{
	case 1: //扩展诊断会话
		bin10.SetAt(1, 0x03);
		bin2701.SetAt(1, 0x01);
		bin2702.SetAt(1, 0x02);
		break;
	case 2: //LEVEL2 编程会话
		bin10.SetAt(1, 0x02);//10 02
		bin2701.SetAt(1, 0x09);//27 09 
		bin2702.SetAt(1, 0x0A);//27 0A 00 00 00 00
		break;
	case 3: //LEVEL2 编程会话-瑞立ECU
		bin10.SetAt(1, 0x02);//10 02
		bin2701.SetAt(1, 0x01);//27 01 
		bin2702.SetAt(1, 0x02);//27 02 00 00 00 00
		break;
	case 4: //LEVEL2 
		bin10.SetAt(1, 0x03);//10 03
		bin2701.SetAt(1, 0x03);//27 03 
		bin2702.SetAt(1, 0x04);//27 04 00 00 00 00
		break;
	default:
		break;
	}

	StatusBox(TEXT_SECURITY_ING);//去掉“TEXT_”后，可以在TEXT文本中查询到

	CCommEcu::SetTimeout(15000);
	CCommEcu::SetCommTime(5, 1, 10000, 50);	//njb 20190620 修改
	binReciver = SendReceiveFrame(g_set.default_flag, bin10);
	if ((binReciver.IsEmpty()) || (binReciver[0] == 0x7F))
	{
		MessageBox(TEXT_SECURITY_FAILED);
		return FALSE;
	}

	//延时1s再执行27解锁
	Delay(1000);

	binReciver = SendReceiveFrame(g_set.default_flag, bin2701);

	if (binReciver.IsEmpty())
	{
		MessageBox(TEXT_COMM_FAIL);
		return FALSE;
	}

	if (binReciver[0] == 0x7F)
	{
		NegativeResponse_0x7F(binReciver);
		return FALSE;
	}

	if (binReciver.GetSize() < 6)
	{
		MessageBox(TEXT_SECURITY_FAILED);
		return false;
	}

	uiSEED = (binReciver[2] << 24) + (binReciver[3] << 16) + (binReciver[4] << 8) + binReciver[5];
	if ((uiSEED == 0) || (uiSEED == 0xFFFFFFFF))
	{
		return TRUE;
	}

	if (!Get_SecurityMask(uiMask))
	{
		MessageBox(TEXT_SECURITY_FAILED);//读取掩码失败
		return FALSE;
	}

	if (iLevel == 1)
	{
		uiKEY = SeedToKey(uiSEED, uiMask);
	}
	else if (iLevel == 2)
	{
		uiKEY = SeedToKey_FBL(uiSEED, uiMask);
	}
	else if (iLevel == 3)
	{
		uiKEY = SeedToKey(uiSEED, uiMask);
	}
	else if (iLevel == 4)
	{
		uiKEY = SeedToKey_Level2(uiSEED, uiMask);
	}


	bin2702[2] = (BYTE)((uiKEY & 0xFF000000) >> 24);
	bin2702[3] = (BYTE)((uiKEY & 0x00FF0000) >> 16);
	bin2702[4] = (BYTE)((uiKEY & 0x0000FF00) >> 8);
	bin2702[5] = (BYTE)((uiKEY & 0x000000FF));

	binReciver = SendReceiveFrame(g_set.default_flag, bin2702);
	Delay(500);

	if (binReciver.IsEmpty())
	{
		MessageBox(TEXT_COMM_FAIL);
		return FALSE;
	}

	if ((binReciver[0] == 0x7F))
	{
		NegativeResponse_0x7F(binReciver);
		MessageBox(TEXT_SECURITY_FAILED);
		return FALSE;
	}

	return TRUE;
}

DWORD SeedToKey_FBL_G050(UINT dwSeed, UINT dwMask)
{
	DWORD dwKey = 0;

	BYTE seed[4];
	BYTE Arrayxor[4];

	seed[0] = (dwSeed >> 24) & 0xff;
	seed[1] = (dwSeed >> 16) & 0xff;
	seed[2] = (dwSeed >> 8) & 0xff;
	seed[3] = (dwSeed >> 0) & 0xff;

	Arrayxor[3] = (dwMask >> 24) & 0xff;
	Arrayxor[2] = (dwMask >> 16) & 0xff;
	Arrayxor[1] = (dwMask >> 8) & 0xff;
	Arrayxor[0] = (dwMask >> 0) & 0xff;

	for (int i = 0; i < 4; i++)
	{
		if (i == 0)
			Arrayxor[i] = Arrayxor[i] ^ 0x35;
		else
			Arrayxor[i] = Arrayxor[i - 1] + (Arrayxor[i] ^ 0x36);
	}

	BYTE val[4] = { 0 };
	BYTE valKey[4] = { 0 };
	val[0] = seed[0] ^ Arrayxor[0];
	val[1] = seed[1] ^ Arrayxor[1];
	val[2] = seed[2] ^ Arrayxor[2];
	val[3] = seed[3] ^ Arrayxor[3];

	valKey[0] = ((val[3] & 0xf0) >> 4) | (16 * (val[0] & 0x0F));
	valKey[1] = ((val[2] & 0xF0) >> 4) | (4 * (val[1] & 0x0F));
	valKey[2] = ((val[2] & 0x3E) >> 1) | (8 * (val[3] & 0x3C));
	//if (g_currIdm == 0x8B0301)
	//	valKey[3] = ((val[1] & 0x0F) >> 4) | (val[2] & 0xF0);
	//else
		valKey[3] = ((val[1] & 0x5F) >> 4) | (val[2] & 0xF0);

	dwKey = (valKey[0] << 24) + (valKey[1] << 16) + (valKey[2] << 8) + valKey[3];

	return dwKey;
}

BOOL Security_WULING_G050(int iLevel)
{
	BOOL	bFlag = TRUE;
	DWORD   uiSEED = 0;
	DWORD   uiMask = 0;
	DWORD   uiKEY = 0;
	CBinary binReciver;

	CBinary bin10("\x10\x03", 2);
	CBinary bin2701("\x27\x01", 2);
	CBinary bin2702("\x27\x02\x00\x00\x00\x00", 6);

	switch (iLevel)
	{
	case 1: //扩展诊断会话
		bin10.SetAt(1, 0x03);
		bin2701.SetAt(1, 0x01);
		bin2702.SetAt(1, 0x02);
		break;
	case 2: //LEVEL_FBL 编程会话
		bin10.SetAt(1, 0x02);//10 02
		bin2701.SetAt(1, 0x09);//27 09 
		bin2702.SetAt(1, 0x0A);//27 0A 00 00 00 00
		break;
	case 3: //LEVEL2 
		bin10.SetAt(1, 0x03);//10 03
		bin2701.SetAt(1, 0x03);//27 03 
		bin2702.SetAt(1, 0x04);//27 04 00 00 00 00
		break;
	default:
		break;
	}

	StatusBox(TEXT_SECURITY_ING);//去掉“TEXT_”后，可以在TEXT文本中查询到

	CCommEcu::SetTimeout(15000);
	CCommEcu::SetCommTime(5, 1, 10000, 50);	
	binReciver = SendReceiveFrame(g_set.default_flag, bin10);
	if ((binReciver.IsEmpty()) || (binReciver[0] == 0x7F))
	{
		MessageBox(TEXT_SECURITY_FAILED);
		return FALSE;
	}

	//延时1s再执行27解锁
	Delay(1000);

	binReciver = SendReceiveFrame(g_set.default_flag, bin2701);

	if (binReciver.IsEmpty())
	{
		MessageBox(TEXT_COMM_FAIL);
		return FALSE;
	}

	if (binReciver[0] == 0x7F)
	{
		NegativeResponse_0x7F(binReciver);
		return FALSE;
	}

	if (binReciver.GetSize() < 6)
	{
		MessageBox(TEXT_SECURITY_FAILED);
		return false;
	}

	uiSEED = (binReciver[2] << 24) + (binReciver[3] << 16) + (binReciver[4] << 8) + binReciver[5];
	if ((uiSEED == 0) || (uiSEED == 0xFFFFFFFF))
	{
		return TRUE;
	}

	if (!Get_SecurityMask(uiMask))
	{
		MessageBox(TEXT_SECURITY_FAILED);//读取掩码失败
		return FALSE;
	}

	if (iLevel == 1)
	{
		uiKEY = SeedToKey_G050(uiSEED, uiMask);
	}
	else if (iLevel == 2)
	{
		uiKEY = (g_currIdm == 0x8B0205 ||g_currIdm== 0x8B0204) ? SeedToKey_FBL(uiSEED, uiMask) : SeedToKey_FBL_G050(uiSEED, uiMask);
	}
	else if (iLevel == 3)
	{
		uiKEY = SeedToKey_Level2_050(uiSEED, uiMask);
	}

	bin2702[2] = (BYTE)((uiKEY & 0xFF000000) >> 24);
	bin2702[3] = (BYTE)((uiKEY & 0x00FF0000) >> 16);
	bin2702[4] = (BYTE)((uiKEY & 0x0000FF00) >> 8);
	bin2702[5] = (BYTE)((uiKEY & 0x000000FF));

	binReciver = SendReceiveFrame(g_set.default_flag, bin2702);
	Delay(500);

	if (binReciver.IsEmpty())
	{
		MessageBox(TEXT_COMM_FAIL);
		return FALSE;
	}

	if ((binReciver[0] == 0x7F))
	{
		NegativeResponse_0x7F(binReciver);
		MessageBox(TEXT_SECURITY_FAILED);
		return FALSE;
	}

	return TRUE;
}

	BYTE val[4] = { 0 };
	BYTE valKey[4] = { 0 };

	seed[0] = (dwSeed >> 24) & 0xff;
	seed[1] = (dwSeed >> 16) & 0xff;
	seed[2] = (dwSeed >> 8) & 0xff;
	seed[3] = (dwSeed >> 0) & 0xff;

	Arrayxor[3] = (dwMask >> 24) & 0xff;
	Arrayxor[2] = (dwMask >> 16) & 0xff;
	Arrayxor[1] = (dwMask >> 8) & 0xff;
	Arrayxor[0] = (dwMask >> 0) & 0xff;

	Temp[0] = seed[0] ^ Arrayxor[1];//临时变量
	Temp[1] = seed[1] + Arrayxor[0];
	Temp[2] = seed[2] ^ Arrayxor[3];
	Temp[3] = seed[3] + (Arrayxor[2] >> 2);
	
	for (int i = 0; i < 4; i++)
	{
		if (i == 0) {
			Arrayxor[i] = Arrayxor[i] ^ Temp[1];
		}
			
		else {
			Arrayxor[i] = Arrayxor[i - 1] + (Arrayxor[i] ^ 0x79);
		}
	    val[i] = seed[i] ^ Arrayxor[i];
	}

//val对应buffer变量，seed对应iSeedArray，Arrayxor对应Mask;
	valKey[0] = ((Temp[0] & 0x0F)) | ((val[1] & 0xF0));

	valKey[2] = ((Temp[3] & 0xAC)) | ((val[0] & 0x53));

	dwKey = (valKey[0] << 24) + (valKey[1] << 16) + (valKey[2] << 8) + valKey[3];
